  ✓ echo - Imprimir texto
  ✓ mkdir - Crear directorios

  ✓ ls - Listar archivos
  ✓ touch - Crear archivos

  ✓ chmod 755, 644 - Formato octal
  ✓ chown - Cambiar propietario
  ✓ ls -l - Ver permisos en detalle

OPERACIONES (Ejercicios 15-25):
  ✓ mkdir -p - Crear directorios anidados
  ✓ cp - Copiar archivos
  ✓ rm - Eliminar archivos
  ✓ mv - Mover/renombrar
  ✓ find - Buscar archivos
  ✓ cat - Ver contenido

AVANZADO (Ejercicios 26-32):
  ✓ chmod -R - Cambios recursivos
  ✓ chgrp - Cambiar grupo
  ✓ ln -s - Crear enlaces
  ✓ ls -lh - Formato legible
  ✓ stat - Información detallada
  ✓ grep - Buscar en contenido
